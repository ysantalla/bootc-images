# Packages added on top of kinoite-main (codecs and Podman are already there).
# One per line. Lines starting with # are ignored.

# System containers and KVM virtual machines
incus
# Terminal: Yakuake is the F12 drop-down, shell.sh holds the rest of the setup
zsh
zsh-autosuggestions
zsh-syntax-highlighting
yakuake

# PDF and document viewer
okular

# Video player. Native, so it uses the image's full ffmpeg and the machine's VA-API driver
haruna

# Same metrics as Calibri and Cambria, so Office documents keep their layout.
# Microsoft's fonts are not redistributable in this public image.
google-carlito-fonts
google-crosextra-caladea-fonts

# Docker commands on top of Podman, with no Docker engine.
# `docker compose` runs the real Compose through Podman's socket.
podman-docker
docker-compose

# Incus VMs: the incus RPM does not pull these in (edk2-ovmf and qemu-img already come from upstream)
qemu-system-x86-core
# Incus always gives VMs a virtio-vga device, and QEMU ships it as a separate module
qemu-device-display-virtio-vga
virtiofsd
gdisk
